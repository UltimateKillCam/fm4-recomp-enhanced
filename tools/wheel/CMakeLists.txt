cmake_minimum_required(VERSION 3.25)
project(FM4WheelTools LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "FM4 wheel tools currently use Windows DirectInput")
endif()
add_library(wheel_host STATIC ../../fm4/src/wheel_host.cpp)
target_link_libraries(wheel_host PUBLIC dinput8 dxguid ole32 user32)
add_executable(wheel-probe ../wheel-probe.cpp)
target_link_libraries(wheel-probe PRIVATE dinput8 dxguid ole32 user32)
add_executable(wheel-tests ../wheel-tests.cpp)
target_link_libraries(wheel-tests PRIVATE wheel_host)
target_compile_options(wheel-tests PRIVATE
  $<$<CXX_COMPILER_ID:MSVC>:/UNDEBUG>
  $<$<NOT:$<CXX_COMPILER_ID:MSVC>>:-UNDEBUG>)
enable_testing()
add_test(NAME wheel-regressions COMMAND wheel-tests)
add_executable(visibility-counter-test visibility-counter-test.cpp)
add_test(NAME cumulative-visibility-counter COMMAND visibility-counter-test)
add_executable(gpu-shader-completion-test gpu-shader-completion-test.cpp)
target_link_libraries(gpu-shader-completion-test PRIVATE d3d12 dxgi)
add_test(NAME gpu-shader-completion-ordering COMMAND gpu-shader-completion-test)
set_tests_properties(gpu-shader-completion-ordering PROPERTIES TIMEOUT 40 SKIP_RETURN_CODE 77)
set(FM4_SDK_INSTALL "" CACHE PATH "Optional pinned SDK install for SDL virtual output tests")
if(FM4_SDK_INSTALL)
  add_executable(hot-stub-test hot-stub-test.cpp)
  set_property(TARGET hot-stub-test PROPERTY CXX_STANDARD 23)
  set_property(TARGET hot-stub-test PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
  target_compile_definitions(hot-stub-test PRIVATE SPDLOG_FMT_EXTERNAL _ITERATOR_DEBUG_LEVEL=0)
  target_include_directories(hot-stub-test PRIVATE "${FM4_SDK_INSTALL}/include")
  target_link_libraries(hot-stub-test PRIVATE "${FM4_SDK_INSTALL}/lib/rexruntime.lib" "${FM4_SDK_INSTALL}/lib/fmt.lib")
  add_test(NAME hot-stub-register-preservation COMMAND hot-stub-test)
  set_tests_properties(hot-stub-register-preservation PROPERTIES TIMEOUT 10
    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${FM4_SDK_INSTALL}/bin")
  add_executable(timing-test timing-test.cpp)
  set_property(TARGET timing-test PROPERTY CXX_STANDARD 23)
  set_property(TARGET timing-test PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
  target_compile_definitions(timing-test PRIVATE SPDLOG_FMT_EXTERNAL _ITERATOR_DEBUG_LEVEL=0)
  target_include_directories(timing-test PRIVATE "${FM4_SDK_INSTALL}/include")
  target_link_libraries(timing-test PRIVATE "${FM4_SDK_INSTALL}/lib/rexruntime.lib")
  add_test(NAME alertable-timing-and-apc COMMAND timing-test)
  set_tests_properties(alertable-timing-and-apc PROPERTIES TIMEOUT 15
    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${FM4_SDK_INSTALL}/bin")
  add_executable(wheel-diagnostic-test diagnostic-test.cpp)
  set_property(TARGET wheel-diagnostic-test PROPERTY CXX_STANDARD 23)
  set_property(TARGET wheel-diagnostic-test PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
  target_compile_definitions(wheel-diagnostic-test PRIVATE SPDLOG_FMT_EXTERNAL _ITERATOR_DEBUG_LEVEL=0)
  target_include_directories(wheel-diagnostic-test PRIVATE "${FM4_SDK_INSTALL}/include")
  target_link_libraries(wheel-diagnostic-test PRIVATE "${FM4_SDK_INSTALL}/lib/rexruntime.lib" "${FM4_SDK_INSTALL}/lib/fmt.lib")
  add_test(NAME wheel-diagnostic-attachment COMMAND wheel-diagnostic-test)
  set_tests_properties(wheel-diagnostic-attachment PROPERTIES TIMEOUT 10
    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${FM4_SDK_INSTALL}/bin")
  add_executable(quiet-window-test quiet-window-test.cpp ../sdk/quiet_window.cpp)
  target_compile_definitions(quiet-window-test PRIVATE SDL_STATIC)
  target_include_directories(quiet-window-test PRIVATE "${FM4_SDK_INSTALL}/include")
  add_test(NAME quiet-window-guards COMMAND quiet-window-test)
  add_executable(controller-virtual-test controller-virtual-test.cpp ../sdk/controller_bridge.cpp)
  # The installed SDK is Release. Match its STL/CRT ABI even when the pure
  # wheel policy assertions are built in Debug.
  set_property(TARGET controller-virtual-test PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
  target_compile_definitions(controller-virtual-test PRIVATE FM4_CONTROLLER_STATIC_TEST SPDLOG_FMT_EXTERNAL _ITERATOR_DEBUG_LEVEL=0)
  target_include_directories(controller-virtual-test PRIVATE "${FM4_SDK_INSTALL}/include")
  target_link_libraries(controller-virtual-test PRIVATE
    "${FM4_SDK_INSTALL}/lib/SDL3-static.lib" "${FM4_SDK_INSTALL}/lib/rexruntime.lib"
    "${FM4_SDK_INSTALL}/lib/fmt.lib"
    kernel32 user32 gdi32 winmm imm32 ole32 oleaut32 version uuid advapi32 setupapi shell32 hid dinput8)
  add_test(NAME controller-virtual-output COMMAND controller-virtual-test)
  set_tests_properties(controller-virtual-output PROPERTIES TIMEOUT 15
    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${FM4_SDK_INSTALL}/bin")
endif()
