# Optional app-local Microsoft Agility runtime for graphics diagnostics. This
# does not install Windows components or change the game's runtime selection.
set(FM4_GRAPHICS_AGILITY_DIR "" CACHE PATH "Private Agility SDK x64 DLL directory")
set(FM4_GRAPHICS_AGILITY_VERSION "" CACHE STRING "Agility D3D12SDKVersion export")
foreach(replay_target neural-scene-replay dlss-scene-replay fsr-scene-replay)
  if(TARGET ${replay_target})
    target_link_libraries(${replay_target} PRIVATE dxguid)
  endif()
endforeach()
if(FM4_GRAPHICS_AGILITY_DIR)
  if(NOT FM4_GRAPHICS_AGILITY_VERSION MATCHES "^[0-9]+$")
    message(FATAL_ERROR "Set FM4_GRAPHICS_AGILITY_VERSION to the SDK's numeric version")
  endif()
  foreach(replay_target neural-scene-replay dlss-scene-replay fsr-scene-replay)
    if(TARGET ${replay_target})
      target_compile_definitions(${replay_target} PRIVATE FM4_GRAPHICS_AGILITY_VERSION=${FM4_GRAPHICS_AGILITY_VERSION})
      foreach(dll D3D12Core.dll d3d12SDKLayers.dll)
        if(NOT EXISTS "${FM4_GRAPHICS_AGILITY_DIR}/${dll}")
          message(FATAL_ERROR "Missing app-local Agility component: ${dll}")
        endif()
        add_custom_command(TARGET ${replay_target} POST_BUILD
          COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:${replay_target}>/D3D12"
          COMMAND ${CMAKE_COMMAND} -E copy_if_different "${FM4_GRAPHICS_AGILITY_DIR}/${dll}"
            "$<TARGET_FILE_DIR:${replay_target}>/D3D12/${dll}")
      endforeach()
    endif()
  endforeach()
endif()
