cmake_minimum_required(VERSION 3.25)
project(fm4_graphics_tools LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(FM4_STREAMLINE "${CMAKE_CURRENT_SOURCE_DIR}/../../.local/streamline-current")
set(FM4_NGX "${FM4_STREAMLINE}/external/ngx-sdk")
if(EXISTS "${FM4_NGX}/include/nvsdk_ngx.h")
  add_library(fm4-neural-forwarder SHARED ../neural/nvngx_forwarder.cpp)
  set_target_properties(fm4-neural-forwarder PROPERTIES OUTPUT_NAME "nvngx.dll_fm4" PREFIX "")
else()
  message(FATAL_ERROR "Neural Rendering bridge needs the NGX SDK headers under ${FM4_NGX}")
endif()
